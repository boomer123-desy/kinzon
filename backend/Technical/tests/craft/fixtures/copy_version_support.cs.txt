namespace Kingzon.OLCM.Base.Services {
 public class CommonService<T> where T:BaseEntity {
  readonly CheckDb db;
  public CommonService(CheckDb db) {this.db=db;}
  public IQueryable<T> GetAll() { var q=db.Set<T>().AsQueryable(); return typeof(ISoftDeletedEntity).IsAssignableFrom(typeof(T))?q.Where(x=>!EF.Property<bool>(x,"IsDeleted")):q; }
  public Task<T?> GetByIdAsync(long id)=>GetAll().FirstOrDefaultAsync(x=>EF.Property<long>(x,"Id")==id);
  public async Task InsertAsync(T entity) { await db.Set<T>().AddAsync(entity); }
  public async Task InsertAsync(List<T> entities) { await db.Set<T>().AddRangeAsync(entities); }
  public void Delete(T entity,bool isSoftDelete=true) {if(isSoftDelete&&entity is ISoftDeletedEntity soft)soft.IsDeleted=true;else db.Remove(entity);}
 }
}

public class Utils {
 public Dictionary<(InventoryDataType,long),DataInfo> Infos {get;}=new();
 public Dictionary<(long,long?),List<GetProductUnitDto>> ProductUnits {get;}=new();
 public DataInfo? GetDataInfo(InventoryDataType type,long id)=>Infos.GetValueOrDefault((type,id));
}
public class ProductService(Utils utils) {
 public List<ProductIdDto> GetProductUnitList(List<ProductIdDto> products) {
  foreach(var product in products) product.UnitList=utils.ProductUnits.GetValueOrDefault((product.ProductId,product.SpecificationId));
  return products;
 }
}
public class CheckDb(DbContextOptions<CheckDb> options):DbContext(options) {
 protected override void OnModelCreating(ModelBuilder model) {
  model.ApplyConfiguration(new ProductionMBomMap()); model.ApplyConfiguration(new ProductionMBomVersionMap());
  model.ApplyConfiguration(new ProductionMBomDetailMap()); model.ApplyConfiguration(new TechnicalCraftMap());
  model.ApplyConfiguration(new TechnicalNodeSettingTemplateMaterialRLMap());
  model.ApplyConfiguration(new TechnicalNodeSettingTemplateMap()); model.ApplyConfiguration(new Technical_NodeTemplateSettingMachineRLMap());
  // Version map refers to host-only InvalidTime/EffectiveTime columns; retain the version uniqueness constraint tested here.
  model.Entity<Technical_CraftOrderVersion>().HasIndex(x=>new{x.CraftOrderId,x.SequenceNo}).IsUnique();
  model.Entity<Technical_CraftOrderVersion>().HasIndex(x=>x.BomVersionId).IsUnique();
  model.Entity<Technical_NodeSettingTemplate>().HasOne(x=>x.NodeTemplate).WithMany().HasForeignKey(x=>x.NodeTemplateId);
  model.Entity<Base_Machine>().HasMany(x=>x.CraftRLs).WithOne().HasForeignKey(x=>x.MachineId);
 }
