namespace Kingzon.OLCM.Base.Services {
 public class CommonService<T>(QueryDb db) where T : BaseEntity {
  public IQueryable<T> GetAll() { var query = db.Set<T>().AsQueryable(); return typeof(ISoftDeletedEntity).IsAssignableFrom(typeof(T)) ? query.Where(x => !EF.Property<bool>(x, "IsDeleted")) : query; }
 }
}
public class QueryDb(DbContextOptions<QueryDb> options) : DbContext(options) {
 protected override void OnModelCreating(ModelBuilder model) {
  model.ApplyConfiguration(new TechnicalCraftMap()); model.ApplyConfiguration(new TechnicalNodeSettingTemplateMap()); model.ApplyConfiguration(new TechnicalNodeSettingTemplateMaterialRLMap());
  // The pre-existing craft-version map references absent timestamp columns; only map its new link here.
  model.Entity<Technical_CraftOrderVersion>().HasIndex(x => x.BomVersionId).IsUnique();
  model.Entity<Base_UnitConversion>();
  model.Entity<Inventory_Record>().HasMany(x => x.OccupyList).WithOne().HasForeignKey(x => x.InventoryRecordId);
 }
}
